ARG CLUB_IMAGE_REGISTRY_PREFIX=public.ecr.aws/docker/library/
FROM ${CLUB_IMAGE_REGISTRY_PREFIX}golang:1.26-bookworm AS go
FROM ${CLUB_IMAGE_REGISTRY_PREFIX}node:22-bookworm-slim AS node

FROM ${CLUB_IMAGE_REGISTRY_PREFIX}eclipse-temurin:21-jdk-noble AS build
RUN apt-get update \
    && apt-get install -y --no-install-recommends git python3 \
    && rm -rf /var/lib/apt/lists/*
COPY --from=go /usr/local/go /usr/local/go
COPY --from=node /usr/local/bin/node /usr/local/bin/node
COPY --from=node /usr/local/lib/node_modules/npm /usr/local/lib/node_modules/npm
RUN ln -s /usr/local/lib/node_modules/npm/bin/npm-cli.js /usr/local/bin/npm
ENV PATH="/usr/local/go/bin:${PATH}"
WORKDIR /src
COPY apps/server/ apps/server/
COPY apps/web/ apps/web/
COPY apps/client-qt/qml/assets/card-back.jpg apps/client-qt/qml/assets/card-back.jpg
COPY apps/client-qt/qml/assets/mana/ apps/client-qt/qml/assets/mana/
COPY deploy/club/ deploy/club/
COPY third_party/forge-runtime/ third_party/forge-runtime/
COPY LICENSE THIRD-PARTY-NOTICES.md ./
RUN bash deploy/club/build.sh /tmp/bundles \
    && mkdir /release \
    && mv /tmp/bundles/*/* /release/ \
    && sed -i 's/-bind 127[.]0[.]0[.]1 /-bind 0.0.0.0 /' /release/run.sh

FROM ${CLUB_IMAGE_REGISTRY_PREFIX}eclipse-temurin:21-jre-noble
RUN useradd --system --user-group --uid 10001 --home-dir /var/lib/hexproof-club --shell /usr/sbin/nologin club \
    && install -d -o club -g club -m 0700 /var/lib/hexproof-club
WORKDIR /opt/hexproof-club
COPY --from=build /release/ ./
ENV CLUB_BIND=0.0.0.0 \
    CLUB_PORT=10000 \
    CLUB_STATE_DIR=/var/lib/hexproof-club \
    CLUB_FORGE_HEAP_MIB=512 \
    JAVA_TOOL_OPTIONS="-Xms32m -XX:+UseSerialGC -XX:ActiveProcessorCount=1"
USER club
EXPOSE 10000
ENTRYPOINT ["/bin/bash", "-c", "CLUB_PORT=\"${PORT:-$CLUB_PORT}\" exec /opt/hexproof-club/run.sh"]
